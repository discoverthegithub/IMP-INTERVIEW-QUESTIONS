Overloading vs Overriding (static vs dynamic binding)
// OVERLOADING — same name, diff params, COMPILE TIME (static/early binding)
int    add(int a, int b)       { return a+b; }
double add(double a, double b) { return a+b; }
void   print(int x)  { cout<<x; }
void   print(string x){ cout<<x; }

// OVERRIDING — child redefines parent, RUNTIME (dynamic/late binding)
class Animal {
  public: virtual void speak() { cout<<"..."; }
};
class Dog : public Animal {
  public: void speak() override { cout<<"Woof"; }
};
Animal* a = new Dog();
a->speak(); // "Woof" — resolved at RUNTIME via vtable
Key line: "Overloading = static binding (compile-time). Overriding = dynamic binding (runtime via virtual functions/vtable)."

Constructor vs Destructor
class Student {
  int* scores;
  public:
    Student(int n) {           // Constructor - called on object creation
      scores = new int[n];
      cout << "Created\n";
    }
    ~Student() {               // Destructor - called on object destruction
      delete[] scores;         // free dynamic memory
      cout << "Destroyed\n";
    }
    Student(const Student& s) { // Copy constructor
      scores = new int[10];
      // deep copy — own memory, not shared pointer
    }
};
// Order: Constructor runs first. Destructor runs when scope ends or delete called.

// 1. Reverse array — two pointer O(n) O(1)
void reverse(int a[], int n) {
    int l=0, r=n-1;
    while(l<r) { swap(a[l++], a[r--]); }
}

// 2. Second largest O(n) O(1)
int secondLargest(int a[], int n) {
    int first=INT_MIN, second=INT_MIN;
    for(int i=0;i<n;i++) {
        if(a[i]>first) { second=first; first=a[i]; }
        else if(a[i]>second && a[i]!=first) second=a[i];
    }
    return second;
}

// 3. Remove duplicates (sorted) O(n) O(1)
int removeDuplicates(int a[], int n) {
    int j=0;
    for(int i=1;i<n;i++)
        if(a[i]!=a[j]) a[++j]=a[i];
    return j+1; // new length
}

// 4. Two Sum — HashMap O(n) O(n)
#include <unordered_map>
pair<int,int> twoSum(int a[], int n, int target) {
    unordered_map<int,int> seen;
    for(int i=0;i<n;i++) {
        int comp=target-a[i];
        if(seen.count(comp)) return {seen[comp],i};
        seen[a[i]]=i;
    }
    return {-1,-1};
}

// Palindrome check O(n) O(1)
bool isPalindrome(string s) {
    int l=0, r=s.size()-1;
    while(l<r) if(s[l++]!=s[r--]) return false;
    return true;
}

// Frequency count O(n) O(k)
void freq(string s) {
    unordered_map<char,int> mp;
    for(char c:s) mp[c]++;
    for(auto& p:mp) cout<<p.first<<":"<<p.second<<"\n";
}
Binary Search O(log n)
int binarySearch(int a[], int n, int target) {
    int l=0, r=n-1;
    while(l<=r) {
        int mid=l+(r-l)/2;  // avoids overflow vs (l+r)/2
        if(a[mid]==target) return mid;
        else if(a[mid]<target) l=mid+1;
        else r=mid-1;
    }
    return -1;
}
Linked List — fast-slow pointer
struct Node { int data; Node* next; };

// Detect cycle — Floyd's algorithm O(n) O(1)
bool hasCycle(Node* head) {
    Node *slow=head, *fast=head;
    while(fast && fast->next) {
        slow=slow->next; fast=fast->next->next;
        if(slow==fast) return true;
    }
    return false;
}

// Reverse linked list O(n) O(1)
Node* reverse(Node* head) {
    Node *prev=nullptr, *curr=head, *next;
    while(curr) {
        next=curr->next;
        curr->next=prev;
        prev=curr; curr=next;
    }
    return prev;
}
Stack — sort using 2nd stack
// Sort stack using extra stack O(n²) O(n)
void sortStack(stack<int>& s) {
    stack<int> temp;
    while(!s.empty()) {
        int curr=s.top(); s.pop();
        while(!temp.empty() && temp.top()>curr) {
            s.push(temp.top()); temp.pop();
        }
        temp.push(curr);
    }
    while(!temp.empty()) { s.push(temp.top()); temp.pop(); }
}

// BFS — uses queue, level-order, shortest path unweighted
void bfs(vector<vector<int>>& adj, int src, int n) {
    vector<bool> vis(n,false);
    queue<int> q;
    vis[src]=true; q.push(src);
    while(!q.empty()) {
        int u=q.front(); q.pop(); cout<<u<<" ";
        for(int v:adj[u]) if(!vis[v]) { vis[v]=true; q.push(v); }
    }
}
// DFS — uses stack/recursion, explore deep first
void dfs(vector<vector<int>>& adj, int u, vector<bool>& vis) {
    vis[u]=true; cout<<u<<" ";
    for(int v:adj[u]) if(!vis[v]) dfs(adj,v,vis);
}

2nd / 3rd Highest Salary
-- 2nd highest (subquery method)
SELECT MAX(salary) FROM employees
WHERE salary < (SELECT MAX(salary)
                FROM employees);

-- Nth highest — LIMIT + OFFSET
SELECT DISTINCT salary
FROM employees
ORDER BY salary DESC
LIMIT 1 OFFSET 1; -- 2nd (offset 0=1st)
-- OFFSET 2 = 3rd highest

-- Above average salary
SELECT name, salary FROM employees
WHERE salary > (SELECT AVG(salary)
                FROM employees);
GROUP BY + HAVING + Dates
-- Count employees per dept
SELECT dept_id, COUNT(*) AS total
FROM employees
GROUP BY dept_id
HAVING COUNT(*) > 5;

-- Max salary per dept
SELECT dept, MAX(salary)
FROM employees
GROUP BY dept;

-- Last 30 days
SELECT * FROM orders
WHERE created_at >=
  DATE_SUB(NOW(), INTERVAL 30 DAY);
WHERE vs HAVING: WHERE filters rows BEFORE GROUP BY. HAVING filters AFTER. Cannot use aggregate (COUNT, SUM, AVG) in WHERE.  |  DISTINCT: removes duplicate rows. Use in Nth highest so ties don't mess up offset.
Correlated Subquery (ADVANCED — may appear)
-- Find employee who earns more than avg of their OWN dept (not global avg)
SELECT e1.name, e1.salary, e1.dept
FROM employees e1
WHERE e1.salary > (
    SELECT AVG(e2.salary) FROM employees e2
    WHERE e2.dept = e1.dept  -- references outer query's row
);
-- Runs once PER ROW of outer query — that's what makes it "correlated"

Deadlock — 4 conditions (ALL must hold)
  • Mutual Exclusion — resource held by only one process
  • Hold & Wait — holds resource while waiting for another
  • No Preemption — resource can't be forcibly taken
  • Circular Wait — P1→P2→P3→P1 dependency cycle
Break ANY ONE condition = prevent deadlock. Common fix: enforce global lock-acquisition order (break circular wait).

Paging + Virtual Memory
  • Virtual memory — each process sees its own large address space; OS maps to physical RAM via page table
  • Page fault — requested page not in RAM → OS loads from disk (slow)
  • Thrashing — more time page-faulting than working (too many processes, too little RAM)
  • LRU page replacement = evict least recently used page
